The following JSON syntax specifies the AI Module Profiles.

{

  “$schema”: “http://json-schema.org/draft/2020-12/schema”,

  “$id”: “https://schemas.mpai.community/PRF/V2.0/data/AIMProfiles.json”,

  “title”: “AIMProfiles”,

  “type”: “object”,

  “properties”: {

    “Header”: {

      “type”: “string”,

      “pattern”: “^PRF-V[0-9]{1,2}[.][0-9]{1,2}$”

    },

    “Profile”: {

      “type”: “string”,

      “pattern”: “^(ALL|NUL)([+-](AUO|AVG|AVS|AVM|BDO|BDD|FCI|FCO|FCD|EPS| LGP|MEM|SPM|OII|POV|PAV|SPD|SPI|SPO|TXD|TXO|TXR|TRN|VIO)|@EPS(#(PST|PSS|PSF|PSB))+|@TRN(#([a-z]{3})(<->|->)([a-z]{3}))+)+$”

    }

  }

}
